contract OracleDelivery {

    event ProcedureInvoked(string order_id);
    event FeeComputed(uint256 value);
    event Debited(string party, uint256 amount, string currency);
    event Credited(string party, uint256 amount, string currency);
    event OracleDeliveryReconciled(string order_id);

    mapping(bytes32 => Status) public statusOf;
    mapping(bytes32 => mapping(string => int256)) public net;  // key -> ledger -> net
    mapping(bytes32 => uint256) public debitTotal;             // key -> total debited
    mapping(bytes32 => uint256) public creditTotal;            // key -> total credited
    IThresholdQuorumAcceptance public immutable acceptanceGuard;

    constructor(address acceptanceGuard_) {
        require(acceptanceGuard_ != address(0), "guard required");
        acceptanceGuard = IThresholdQuorumAcceptance(acceptanceGuard_);
    }

    function execute(uint256 u_amount, uint256 u_rate, string calldata u_order_id, string calldata u_payer, string calldata u_payee, string calldata u_currency) external {
        bytes32 _k = keccak256(bytes(u_order_id));
        require(statusOf[_k] == Status.None, "replay");
        require(acceptanceGuard.isAccepted(_k), "quorum not accepted");
        emit ProcedureInvoked(u_order_id);

        uint256 u_fee = (u_amount + u_rate);
        emit FeeComputed(u_fee);

        net[_k]["0"] -= int256(u_amount);
        debitTotal[_k] += u_amount;
        emit Debited(u_payer, u_amount, u_currency);
        net[_k]["1"] += int256(u_amount);
        creditTotal[_k] += u_amount;
        emit Credited(u_payee, u_amount, u_currency);

        require(debitTotal[_k] == creditTotal[_k], "not balanced");
        statusOf[_k] = Status.Reconciled;
        emit OracleDeliveryReconciled(u_order_id);
    }
}

/// Generated by the Neutrino Translator from verified MLIR.
/// COORDINATION CONTRACT — not the source of truth and not fund custody.
/// It records coordination checkpoints (status, attestations, net positions)
/// that participants accept; authoritative state stays in their own systems.
/// All coordination state is keyed by the workflow key (the idempotency key),
/// so distinct workflow instances never share status, totals, or net positions.
contract GuardedPair {
    enum Status { None, Reconciled }

    event ProcedureInvoked(string pair_id);
    event ReserveDebitDebited(string party, uint256 amount, string currency);
    event HoldingCreditCredited(string party, uint256 amount, string currency);
    event BufferDebitDebited(string party, uint256 amount, string currency);
    event SettledCreditCredited(string party, uint256 amount, string currency);
    event GuardedPairReconciled(string pair_id);

    mapping(bytes32 => Status) public statusOf;
    mapping(bytes32 => bool) public attested;
    mapping(bytes32 => mapping(string => int256)) public net;  // key -> ledger -> net
    mapping(bytes32 => uint256) public debitTotal;             // key -> total debited
    mapping(bytes32 => uint256) public creditTotal;            // key -> total credited

    function attest(string calldata u_pair_id) external {
        bytes32 _k = keccak256(bytes(u_pair_id));
        require(statusOf[_k] == Status.None, "bad transition");
        attested[_k] = true;
    }

    function execute(string calldata u_pair_id, string calldata u_source_party, string calldata u_target_party, uint256 u_amount, string calldata u_currency, uint256 u_approval, uint256 u_floor_amount) external {
        bytes32 _k = keccak256(bytes(u_pair_id));
        require(statusOf[_k] == Status.None, "replay");
        require(attested[_k], "missing attestation");
        emit ProcedureInvoked(u_pair_id);


        if (u_approval == 1) {
            net[_k]["source.reserve"] -= int256(u_amount);
            debitTotal[_k] += u_amount;
            emit ReserveDebitDebited(u_source_party, u_amount, u_currency);
        }
        if (u_approval == 1) {
            net[_k]["target.holding"] += int256(u_amount);
            creditTotal[_k] += u_amount;
            emit HoldingCreditCredited(u_target_party, u_amount, u_currency);
        }
        if (u_amount >= u_floor_amount) {
            net[_k]["source.buffer"] -= int256(u_amount);
            debitTotal[_k] += u_amount;
            emit BufferDebitDebited(u_source_party, u_amount, u_currency);
        }
        if (u_amount >= u_floor_amount) {
            net[_k]["target.settled"] += int256(u_amount);
            creditTotal[_k] += u_amount;
            emit SettledCreditCredited(u_target_party, u_amount, u_currency);
        }

        require(debitTotal[_k] == creditTotal[_k], "not balanced");
        statusOf[_k] = Status.Reconciled;
        emit GuardedPairReconciled(u_pair_id);
    }
}

interface IThresholdQuorumAcceptance {
    function isAccepted(bytes32 claimId) external view returns (bool);
    function acceptedExecutionClaim(bytes32 claimId) external view returns (bytes32);
}

/// Generated by the Neutrino Translator from verified MLIR.
/// COORDINATION CONTRACT — not the source of truth and not fund custody.
/// It records coordination checkpoints (status, commit ledger, net positions)
/// that participants accept; authoritative state stays in their own systems.
/// All coordination state is keyed by the workflow key (the idempotency key),
/// so distinct workflow instances never share status, commit ledger, or net positions.
contract AcceptedMultiPolicy {
    enum Status { None, Reconciled }

    event ProcedureInvoked(string settlement_id);
    event AlphaDebitDebited(string party, uint256 amount, string currency);
    event BetaDebitDebited(string party, uint256 amount, string currency);
    event AlphaCreditCredited(string party, uint256 amount, string currency);
    event BetaCreditCredited(string party, uint256 amount, string currency);
    event AcceptedMultiPolicyReconciled(string settlement_id);

    mapping(bytes32 => Status) public statusOf;
    mapping(bytes32 => bytes32) public keyClaim;                         // key -> bound execution claim ()
    mapping(bytes32 => mapping(uint256 => bool)) public committedGroup;  // key -> group ordinal -> committed ()
    mapping(bytes32 => mapping(string => int256)) public net;            // key -> ledger -> net
    mapping(bytes32 => uint256) public debitTotal;                       // key -> total debited
    mapping(bytes32 => uint256) public creditTotal;                      // key -> total credited
    IThresholdQuorumAcceptance public immutable acceptanceGuard_alpha_proof;
    IThresholdQuorumAcceptance public immutable acceptanceGuard_beta_proof;

    constructor(address acceptanceGuard_alpha_proof_, address acceptanceGuard_beta_proof_) {
        require(acceptanceGuard_alpha_proof_ != address(0), "guard required");
        acceptanceGuard_alpha_proof = IThresholdQuorumAcceptance(acceptanceGuard_alpha_proof_);
        require(acceptanceGuard_beta_proof_ != address(0), "guard required");
        acceptanceGuard_beta_proof = IThresholdQuorumAcceptance(acceptanceGuard_beta_proof_);
    }

    function execute(string calldata u_settlement_id, string calldata u_payer_id, string calldata u_payee_id, uint256 u_amount, string calldata u_currency) external {
        bytes32 _k = keccak256(bytes(u_settlement_id));
        require(statusOf[_k] != Status.Reconciled, "replay");
        string[] memory _n = new string[](5);
        string[] memory _c = new string[](5);
        string[] memory _v = new string[](5);
        _n[0] = "amount";
        _c[0] = "money";
        _v[0] = CoordinationEnvelope.u2s(u_amount);
        _n[1] = "currency";
        _c[1] = "currency";
        _v[1] = u_currency;
        _n[2] = "payee_id";
        _c[2] = "party";
        _v[2] = u_payee_id;
        _n[3] = "payer_id";
        _c[3] = "party";
        _v[3] = u_payer_id;
        _n[4] = "settlement_id";
        _c[4] = "string";
        _v[4] = u_settlement_id;
        bytes32 _claim = CoordinationEnvelope.claim("d76de55a0346b215373259cbe19fdead41b3d87f51ad7b0243a6b500d97e0a74", u_settlement_id, _n, _c, _v);
        require(keyClaim[_k] == bytes32(0) || keyClaim[_k] == _claim, "claim mismatch");
        emit ProcedureInvoked(u_settlement_id);

        bool _progress = false;

        if ((acceptanceGuard_alpha_proof.acceptedExecutionClaim(_k) == _claim) && !committedGroup[_k][0]) {
            net[_k]["payer.alpha"] -= int256(u_amount);
            debitTotal[_k] += u_amount;
            emit AlphaDebitDebited(u_payer_id, u_amount, u_currency);
            net[_k]["payee.alpha"] += int256(u_amount);
            creditTotal[_k] += u_amount;
            emit AlphaCreditCredited(u_payee_id, u_amount, u_currency);
            committedGroup[_k][0] = true;
            _progress = true;
        }
        if ((acceptanceGuard_beta_proof.acceptedExecutionClaim(_k) == _claim) && !committedGroup[_k][1]) {
            net[_k]["payer.beta"] -= int256(u_amount);
            debitTotal[_k] += u_amount;
            emit BetaDebitDebited(u_payer_id, u_amount, u_currency);
            net[_k]["payee.beta"] += int256(u_amount);
            creditTotal[_k] += u_amount;
            emit BetaCreditCredited(u_payee_id, u_amount, u_currency);
            committedGroup[_k][1] = true;
            _progress = true;
        }
        if (_progress && keyClaim[_k] == bytes32(0)) {
            keyClaim[_k] = _claim;
        }

        bool _done = committedGroup[_k][0] && committedGroup[_k][1];
        if (_done) {
            require(debitTotal[_k] == creditTotal[_k], "not balanced");
            statusOf[_k] = Status.Reconciled;
            emit AcceptedMultiPolicyReconciled(u_settlement_id);
        }
    }
}
